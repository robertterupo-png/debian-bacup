#!/bin/sh
# bookmark.sh — save current Firefox URL into ~/bookmarks.txt
# Dependencies: xdotool, xclip, notify

xdotool key ctrl+l
sleep 0.1
xdotool key ctrl+c
sleep 0.1
xdotool key F6
sleep 0.1

clipboard=$(xclip -o | tr -d '\n' | sed 's/[[:space:]]*$//')
bookmark_file="$HOME/bookmarks.txt"

if [ -z "$clipboard" ]; then
    notify "Error: Clipboard empty"
    exit 1
fi

if grep -Fxq "$clipboard" "$bookmark_file"; then
    notify "Duplicate: Already bookmarked"
else
    printf '%s\n' "$clipboard" >> "$bookmark_file"
    sort -u "$bookmark_file" -o "$bookmark_file"
    notify "Bookmarked: $clipboard"
fi
